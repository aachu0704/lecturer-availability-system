    function appendLog(type, text) {
      const term = document.getElementById('telemetryTerminal');
      if (!term) return;
      const now = new Date();
      const timeStr = now.toTimeString().split(' ')[0] + '.' + String(now.getMilliseconds()).padStart(3, '0');
      const logItem = document.createElement('div');
      logItem.className = 'flex items-start gap-2';

      let tagColor = 'text-brand-600';
      if (type === 'WARN') tagColor = 'text-amber-600';
      if (type === 'ERROR' || type === 'DELETE') tagColor = 'text-rose-600';
      if (type === 'SUCCESS' || type === 'HEARTBEAT') tagColor = 'text-emerald-600';

      logItem.innerHTML = `
        <span class="text-slate-400 shrink-0">[${timeStr}]</span>
        <span class="${tagColor} font-semibold shrink-0">[${type}]</span>
        <span>${text}</span>
      `;
      term.appendChild(logItem);
      term.scrollTop = term.scrollHeight;
    }

    function showToast(title, message, isError = false) {
      const toast = document.getElementById('toastNotification');
      const toastTitle = document.getElementById('toastTitle');
      const toastMsg = document.getElementById('toastMessage');
      const toastIcon = document.getElementById('toastIcon');

      toastTitle.textContent = title;
      toastMsg.textContent = message;
      
      if (isError) {
        toastIcon.textContent = 'error';
        toastIcon.className = 'material-symbols-outlined text-rose-400 text-[20px]';
      } else {
        toastIcon.textContent = 'check_circle';
        toastIcon.className = 'material-symbols-outlined text-emerald-400 text-[20px]';
      }

      toast.classList.remove('translate-y-20', 'opacity-0', 'pointer-events-none');
      setTimeout(() => {
        toast.classList.add('translate-y-20', 'opacity-0', 'pointer-events-none');
      }, 3500);
    }

    function handleStatusChange(selectElem, name) {
      const val = selectElem.value;
      // Reset color classes
      selectElem.className = 'status-select rounded-md border border-slate-200 font-semibold text-xs py-1 px-2.5 focus:outline-none focus:ring-1 focus:ring-brand-500 cursor-pointer ';
      
      if (val === 'Available') {
        selectElem.className += 'bg-emerald-50 text-emerald-800';
      } else if (val === 'Busy') {
        selectElem.className += 'bg-amber-50 text-amber-800';
      } else if (val === 'In Lecture') {
        selectElem.className += 'bg-sky-50 text-sky-800';
      } else {
        selectElem.className += 'bg-rose-50 text-rose-800';
      }

      appendLog('PRESENCE-SYNC', `Admin adjusted status for ${name} -> ${val.toUpperCase()}`);
      showToast('Status Updated', `${name} is marked as ${val}. ESP32 display refreshed.`);
    }

    function resetNode(mac, lecturerName) {
      appendLog('SYNC', `Manual framebuffer refresh dispatched to ${mac} (${lecturerName})`);
      showToast('Hardware Synchronized', `Dispatched refresh frame to node ${mac}.`);
    }

    function deleteLecturer(buttonElement, name) {
      if (confirm(`Remove ${name} from directory and decouple the associated ESP32 display?`)) {
        const row = buttonElement.closest('tr');
        row.remove();
        updateFacultyCount();
        appendLog('DELETE', `Faculty member ${name} and bound hardware node dissociated.`);
        showToast('Faculty Removed', `${name} has been removed.`);
      }
    }

      if (label) {
        label.textContent = `Showing ${visibleRows.length} of ${totalRows.length} registered faculty entries`;
      }
      const kpi = document.getElementById('kpiTotalCount');
      if (kpi) {
        kpi.textContent = totalRows.length;
      }
    }

<script>
    let activeDept = 'all';
    let activeStatus = 'all';

    const searchInput = document.getElementById('faculty-search-input');
    const facultyCards = document.querySelectorAll('.faculty-card');
    const deptChips = document.querySelectorAll('.dept-chip');
    const statusBtns = document.querySelectorAll('.status-btn');

    function applyFilters() {
      const query = (searchInput.value || '').toLowerCase().trim();

      facultyCards.forEach(card => {
        const name = (card.dataset.name || '').toLowerCase();
        const dept = (card.dataset.dept || '');
        const status = (card.dataset.status || '');
        const room = (card.dataset.room || '').toLowerCase();

        const matchesQuery = !query || name.includes(query) || room.includes(query) || dept.toLowerCase().includes(query);
        const matchesDept = (activeDept === 'all') || (dept === activeDept);
        const matchesStatus = (activeStatus === 'all') || (status === activeStatus);

        if (matchesQuery && matchesDept && matchesStatus) {
          card.classList.remove('hidden');
        } else {
          card.classList.add('hidden');
        }
      });
    }

    searchInput.addEventListener('input', applyFilters);

    window.addEventListener('keydown', (e) => {
      if (e.key === '/' && document.activeElement !== searchInput) {
        e.preventDefault();
        searchInput.focus();
        searchInput.select();
      } else if (e.key === 'Escape') {
        closeMapModal();
        closeQueueModal();
        closeNoticeModal();
        closeScheduleModal();
      }
    });

    deptChips.forEach(chip => {
      chip.addEventListener('click', () => {
        deptChips.forEach(c => {
          c.classList.remove('bg-slate-900', 'text-white');
          c.classList.add('bg-slate-100', 'text-slate-600');
        });
        chip.classList.add('bg-slate-900', 'text-white');
        chip.classList.remove('bg-slate-100', 'text-slate-600');

        activeDept = chip.dataset.dept;
        applyFilters();
      });
    });

    statusBtns.forEach(btn => {
      btn.addEventListener('click', () => {
        statusBtns.forEach(b => {
          b.classList.remove('bg-brand-600', 'text-white');
          b.classList.add('bg-slate-100', 'text-slate-600');
        });
        btn.classList.add('bg-brand-600', 'text-white');
        btn.classList.remove('bg-slate-100', 'text-slate-600');

        activeStatus = btn.dataset.filter;
        applyFilters();
      });
    });

    function openMapModal(building, room, name) {
      document.getElementById('modal-map-building').innerText = building;
      document.getElementById('modal-map-room').innerText = room + ' • ' + name;
      document.getElementById('map-modal').classList.remove('hidden');
    }
    function closeMapModal() {
      document.getElementById('map-modal').classList.add('hidden');
    }

    function openQueueModal(facultyName, currentQueueSize) {
      document.getElementById('queue-faculty-name').innerText = facultyName;
      document.getElementById('queue-slot-text').innerText = 'Position #' + (currentQueueSize + 1);
      document.getElementById('queue-modal').classList.remove('hidden');
    }
    function closeQueueModal() {
      document.getElementById('queue-modal').classList.add('hidden');
    }
    function confirmQueueJoin() {
      closeQueueModal();
      showToast('Queue Registered', 'Your queue spot has been reserved. You will receive an SMS reminder.');
    }

    function openNoticeModal(facultyName) {
      document.getElementById('notice-faculty-name').innerText = facultyName;
      document.getElementById('notice-modal').classList.remove('hidden');
    }
    function closeNoticeModal() {
      document.getElementById('notice-modal').classList.add('hidden');
    }
    function sendNoticeAction() {
      closeNoticeModal();
      showToast('Notice Delivered', 'Your message has been routed to the office desk display.');
    }

    function openScheduleModal(facultyName, room) {
      document.getElementById('schedule-faculty-name').innerText = facultyName + ' (' + room + ')';
      document.getElementById('schedule-modal').classList.remove('hidden');
    }
    function closeScheduleModal() {
      document.getElementById('schedule-modal').classList.add('hidden');
    }
    function confirmScheduleAppointment() {
      closeScheduleModal();
      showToast('Appointment Confirmed', 'Office visit added to your university calendar.');
    }

    let toastTimeout;
    function showToast(title, msg) {
      const toast = document.getElementById('live-toast');
      document.getElementById('toast-title').innerText = title;
      document.getElementById('toast-message').innerText = msg;
      toast.classList.remove('hidden');
      clearTimeout(toastTimeout);
      toastTimeout = setTimeout(() => {
        toast.classList.add('hidden');
      }, 4000);
    }
    function hideToast() {
      document.getElementById('live-toast').classList.add('hidden');
    }
  </script>

<script>
    function updateHardwareSimulator(name, room, status, statusColorClass, detail) {
      const syncInd = document.getElementById('live-sync-indicator');
      const tickerFeed = document.getElementById('ticker-feed');

      if (syncInd) {
        syncInd.innerText = 'Transmitting Payload...';
        syncInd.className = 'text-blue-600 font-semibold';
        setTimeout(() => {
          syncInd.innerText = 'Synchronized';
          syncInd.className = 'text-emerald-700 font-semibold';
        }, 1500);
      }

      if (tickerFeed) {
        tickerFeed.innerText = name + ' (' + room + ') -> ' + status + ' • Just now';
      }
    }

    let toastTimer;
    function showNotification(msg) {
      const toast = document.getElementById('toast-notification');
      const msgEl = document.getElementById('toast-message');
      if (!toast || !msgEl) return;

      msgEl.innerText = msg;
      toast.classList.remove('translate-y-24', 'opacity-0', 'pointer-events-none');
      toast.classList.add('translate-y-0', 'opacity-100');

      clearTimeout(toastTimer);
      toastTimer = setTimeout(hideNotification, 3000);
    }

    function hideNotification() {
      const toast = document.getElementById('toast-notification');
      if (!toast) return;
      toast.classList.remove('translate-y-0', 'opacity-100');
      toast.classList.add('translate-y-24', 'opacity-0', 'pointer-events-none');
    }
  </script>
